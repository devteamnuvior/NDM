const BOOK_INTERESTS = ['Wrinkles & fine lines', 'Volume loss', 'Sagging skin & jowls', 'Dark circles & under-eye hollows', 'Double chin', 'Acne', 'Acne scars', 'Pigmentation & sun spots', 'Aptos Thread Lift', 'Dermal Fillers', 'Laser Hair Removal', 'Microneedling', 'Not sure, I’d like a skin analysis'];
const BOOK_TIMES = ['Weekday morning', 'Weekday afternoon', 'Weekday evening', 'Saturday'];
const BOOK_STEPS = ['Your concern', 'Clinic and time', 'Your details'];
function BookChip({ type = 'radio', name, value, checked, onChange, children }) {
  return (
    <label className="relative inline-flex cursor-pointer">
      <input type={type} name={name} value={value} checked={checked} onChange={onChange} className="peer absolute inset-0 m-0 cursor-pointer opacity-0" />
      <span className="inline-flex min-h-[42px] items-center gap-2 rounded-full border border-ndm-ink/15 bg-white px-4 font-inter text-[14px] leading-[1.3] text-ndm-ink transition-colors peer-hover:border-ndm-ink/40 peer-checked:border-ndm-deep peer-checked:bg-ndm-deep peer-checked:text-white peer-focus-visible:ring-4 peer-focus-visible:ring-ndm-blue/25">{children}</span>
    </label>
  );
}
/* the three-step request form, shared by the booking section and the Book Free Consultation pop-up */
function BookForm({ thanksId }) {
  const box = useRef(null);
  const [step, setStep] = useState(0);
  const [sent, setSent] = useState(false);
  const [err, setErr] = useState('');
  const [f, setF] = useState({ interest: '', clinic: 'Uptown', visit: 'New', times: [], via: ['Call'], news: false });
  const pick = (k) => (e) => { setErr(''); setF((o) => ({ ...o, [k]: e.target.value })); };
  const toggle = (k, v) => { setErr(''); setF((o) => ({ ...o, [k]: o[k].includes(v) ? o[k].filter((x) => x !== v) : [...o[k], v] })); };
  const go = (n) => {
    setErr(''); setStep(n);
    setTimeout(() => { const l = box.current && box.current.querySelector('legend'); if (l) l.focus({ preventScroll: true }); }, 30);
  };
  const next = () => { if (step === 0 && !f.interest) { setErr('Please choose one option.'); return; } go(step + 1); };
  const submit = (e) => { e.preventDefault(); if (!f.via.length) { setErr('Please choose at least one way to reach you.'); return; } setSent(true); };
  const legend = 'font-aeonik text-[clamp(24px,2.2vw,30px)] font-normal leading-[1.15] tracking-[-0.02em] text-ndm-ink outline-none';
  const label = 'mb-2 block font-inter text-[13.5px] font-semibold text-ndm-ink';
  if (sent) return <div className="flex flex-1 items-center justify-center py-6"><BookThanks id={thanksId} onAnother={() => { setSent(false); go(0); }} /></div>;
  return (
    <form ref={box} onSubmit={submit} className="flex flex-1 flex-col">
      <div className="flex items-center justify-between gap-4 font-inter text-[13px] text-ndm-muted">
        <span>Step {step + 1} of {BOOK_STEPS.length}</span><span className="font-medium text-ndm-ink">{BOOK_STEPS[step]}</span>
      </div>
      <div className="mt-2.5 grid grid-cols-3 gap-1.5" aria-hidden="true">
        {BOOK_STEPS.map((s, i) => <span key={s} className={`h-1 rounded-full transition-colors duration-500 ${i <= step ? 'bg-ndm-deep' : 'bg-ndm-ink/10'}`} />)}
      </div>

      {step === 0 && (
        <fieldset className="mt-7">
          <legend tabIndex={-1} className={legend}>What would you like to address?</legend>
          <p className="mt-2 font-inter text-[14.5px] text-ndm-muted">Choose the closest match. You can change it later.</p>
          <div className="mt-5 flex flex-wrap gap-2">
            {BOOK_INTERESTS.map((t) => <BookChip key={t} name="interest" value={t} checked={f.interest === t} onChange={pick('interest')}>{t}</BookChip>)}
          </div>
        </fieldset>
      )}

      {step === 1 && (
        <fieldset className="mt-7">
          <legend tabIndex={-1} className={legend}>Where and when?</legend>
          <div className="mt-5 grid gap-5">
            <div role="group" aria-label="Clinic"><span className={label}>Clinic</span>
              <div className="flex flex-wrap gap-2">
                {[['Uptown', 'Uptown, Yonge & Sheppard'], ['Downtown', 'Downtown, Bay & College'], ['Either', 'Either clinic']].map(([v, t]) => <BookChip key={v} name="clinic" value={v} checked={f.clinic === v} onChange={pick('clinic')}>{t}</BookChip>)}
              </div>
            </div>
            <div role="group" aria-label="Have you visited us before?"><span className={label}>Have you visited us before?</span>
              <div className="flex flex-wrap gap-2">
                {[['New', 'I’m new'], ['Returning', 'I’m returning']].map(([v, t]) => <BookChip key={v} name="visit" value={v} checked={f.visit === v} onChange={pick('visit')}>{t}</BookChip>)}
              </div>
            </div>
            <div role="group" aria-label="Preferred time (optional)"><span className={label}>Preferred time <span className="font-normal text-ndm-muted">(optional, choose any)</span></span>
              <div className="flex flex-wrap gap-2">
                {BOOK_TIMES.map((t) => <BookChip key={t} type="checkbox" name="time" value={t} checked={f.times.includes(t)} onChange={() => toggle('times', t)}>{t}</BookChip>)}
              </div>
            </div>
          </div>
        </fieldset>
      )}

      {step === 2 && (
        <fieldset className="mt-7">
          <legend tabIndex={-1} className={legend}>How can we reach you?</legend>
          <div className="mt-5 grid gap-3 sm:grid-cols-2">
            <label><span className={label}>First name</span><input required name="first" autoComplete="given-name" className={fieldCls} /></label>
            <label><span className={label}>Last name</span><input required name="last" autoComplete="family-name" className={fieldCls} /></label>
            <label><span className={label}>Mobile number</span><input required type="tel" name="phone" autoComplete="tel" className={fieldCls} /></label>
            <label><span className={label}>Email</span><input required type="email" name="email" autoComplete="email" className={fieldCls} /></label>
          </div>
          <div role="group" aria-label="Best way to reach you, choose any" className="mt-5"><span className={label}>Best way to reach you <span className="font-normal text-ndm-muted">(choose any)</span></span>
            <div className="flex flex-wrap gap-2">
              {['Call', 'Text', 'Email'].map((t) => <BookChip key={t} type="checkbox" name="via" value={t} checked={f.via.includes(t)} onChange={() => toggle('via', t)}>{t}</BookChip>)}
            </div>
          </div>
          <p className="mt-5 rounded-[14px] bg-ndm-mist px-4 py-3 font-inter text-[13px] leading-[1.5] text-ndm-muted">Please don’t include medical details here. We use your details only to arrange your consultation.</p>
          <label className="mt-4 flex cursor-pointer items-start gap-3 font-inter text-[13.5px] leading-[1.5] text-ndm-muted">
            <input type="checkbox" checked={f.news} onChange={(e) => setF((o) => ({ ...o, news: e.target.checked }))} className="mt-0.5 size-4 shrink-0 accent-[#0F80C3]" />
            <span>Optional: send me occasional news and offers. You can unsubscribe at any time.</span>
          </label>
        </fieldset>
      )}

      {err && <p role="alert" className="mt-4 font-inter text-[13.5px] text-[#B4351F]">{err}</p>}
      <div className="mt-auto flex flex-wrap items-center gap-3 pt-8">
        {step > 0 && <Btn as="button" type="button" v="line" onClick={() => go(step - 1)}>Back</Btn>}
        {step < 2 ? <Btn as="button" type="button" v="ink" onClick={next}>Continue</Btn> : <Btn as="button" type="submit" v="book" hot>Book Free Consultation</Btn>}
        {step === 2 && <span className="font-inter text-[13px] text-ndm-muted">We contact you to confirm a time.</span>}
      </div>
    </form>
  );
}
