/* play a muted film only while it is on screen; never with reduced motion or data saver */
function useFilm(vid, opts = {}) {
  const [paused, setPaused] = useState(REDUCED);
  const user = useRef(false);
  useEffect(() => {
    const v = vid.current; if (!v) return;
    v.muted = true;
    if (REDUCED || (navigator.connection && navigator.connection.saveData)) { v.pause(); setPaused(true); return; }
    const io = new IntersectionObserver(([e]) => {
      if (e.isIntersecting && !user.current) { const p = v.play(); if (p && p.catch) p.catch(() => {}); setPaused(false); }
      else if (!e.isIntersecting) v.pause();
    }, { threshold: 0.15 });
    io.observe(v);
    return () => io.disconnect();
  }, []);
  const toggle = () => {
    const v = vid.current; if (!v) return;
    if (v.paused) { user.current = false; const p = v.play(); if (p && p.catch) p.catch(() => {}); setPaused(false); }
    else { user.current = true; v.pause(); setPaused(true); }
  };
  return [paused, toggle];
}
