const BOOK_INTERESTS = ['Wrinkles & fine lines', 'Volume loss', 'Sagging skin & jowls', 'Dark circles & under-eye hollows', 'Double chin', 'Acne', 'Acne scars', 'Pigmentation & sun spots', 'Aptos Thread Lift', 'Dermal Fillers', 'Laser Hair Removal', 'Microneedling', 'Not sure, I’d like a skin analysis'];
const BOOK_STEPS = ['Your concern', 'Clinic, day and time', 'Your details'];
const BOOK_CLINIC_OPTS = [['Uptown', 'Uptown, Yonge & Sheppard'], ['Downtown', 'Downtown, Bay & College']];
/* opening hours (HOURS): Monday to Friday 9 to 7, Saturday 10 to 5, Sunday closed; a request slot every hour, the last one an hour before closing */
const bookSlots = (d) => { const w = d.getDay(); if (w === 0) return []; const [a, b] = w === 6 ? [10, 16] : [9, 18]; return Array.from({ length: b - a + 1 }, (_, i) => a + i); };
const bookHour = (h) => `${h > 12 ? h - 12 : h}:00 ${h >= 12 ? 'pm' : 'am'}`;
const bookDay = (d) => d.toLocaleDateString('en-CA', { weekday: 'short', month: 'short', day: 'numeric' });
function BookChip({ type = 'radio', name, value, checked, onChange, children }) {
  return (
    <label className="relative inline-flex cursor-pointer">
      <input type={type} name={name} value={value} checked={checked} onChange={onChange} className="peer absolute inset-0 m-0 cursor-pointer opacity-0" />
      <span className="inline-flex min-h-[40px] items-center gap-2 rounded-full border border-ndm-ink/15 bg-white px-3.5 font-inter text-[13.5px] leading-[1.3] text-ndm-ink transition-colors peer-hover:border-ndm-ink/40 peer-checked:border-ndm-deep peer-checked:bg-ndm-deep peer-checked:text-white peer-focus-visible:ring-4 peer-focus-visible:ring-ndm-blue/25">{children}</span>
    </label>
  );
}
/* a month calendar (Sundays and past days closed) and the open request times of the chosen day */
function BookCalendar({ day, time, onDay, onTime }) {
  const today = new Date(); today.setHours(0, 0, 0, 0);
  const [view, setView] = useState(() => new Date(today.getFullYear(), today.getMonth(), 1));
  const first = (view.getDay() + 6) % 7; /* weeks start on Monday */
  const days = new Date(view.getFullYear(), view.getMonth() + 1, 0).getDate();
  const cells = [...Array(first).fill(null), ...Array.from({ length: days }, (_, i) => new Date(view.getFullYear(), view.getMonth(), i + 1))];
  const canBack = view > new Date(today.getFullYear(), today.getMonth(), 1);
  const nav = 'grid size-8 place-items-center rounded-full border border-ndm-ink/12 text-ndm-ink transition-colors hover:bg-ndm-mist disabled:opacity-30';
  return (
    <div className="grid gap-4 md:grid-cols-[minmax(0,1fr)_minmax(0,200px)]">
      <div>
        <div className="flex items-center justify-between">
          <p className="font-aeonik text-[19px] text-ndm-ink">{view.toLocaleDateString('en-CA', { month: 'long', year: 'numeric' })}</p>
          <span className="flex gap-1.5">
            <button type="button" disabled={!canBack} onClick={() => setView(new Date(view.getFullYear(), view.getMonth() - 1, 1))} aria-label="Previous month" className={nav}><IconArrow size={12} className="rotate-180" /></button>
            <button type="button" onClick={() => setView(new Date(view.getFullYear(), view.getMonth() + 1, 1))} aria-label="Next month" className={nav}><IconArrow size={12} /></button>
          </span>
        </div>
        <div className="mt-2 grid grid-cols-7 gap-1 text-center font-inter text-[11px] uppercase tracking-[.08em] text-ndm-muted" aria-hidden="true">{['Mo', 'Tu', 'We', 'Th', 'Fr', 'Sa', 'Su'].map((d) => <span key={d} className="py-1">{d}</span>)}</div>
        <div className="grid grid-cols-7 gap-1" role="group" aria-label="Choose a day">
          {cells.map((d, i) => {
            if (!d) return <span key={`e${i}`} />;
            const closed = d < today || d.getDay() === 0;
            const on = day && d.getTime() === day.getTime();
            return (
              <button key={d.getDate()} type="button" disabled={closed} onClick={() => onDay(d)} aria-pressed={on} aria-label={`${bookDay(d)}${closed ? ', closed' : ''}`}
                className={`relative grid h-9 place-items-center rounded-[10px] font-inter text-[13.5px] tabular-nums transition-colors ${on ? 'bg-ndm-deep text-white' : closed ? 'text-ndm-ink/25' : 'bg-white text-ndm-ink ring-1 ring-ndm-ink/10 hover:ring-ndm-deep'}`}>
                {d.getDate()}{!closed && !on && <i aria-hidden="true" className="absolute bottom-1 size-1 rounded-full bg-ndm-sky" />}
              </button>
            );
          })}
        </div>
      </div>
      <div>
        <p className="font-inter text-[13px] font-semibold text-ndm-ink">{day ? bookDay(day) : 'Pick a day'}</p>
        {day ? (
          <div className="mt-2 grid grid-cols-3 gap-1.5 md:grid-cols-2" role="group" aria-label="Choose a time">
            {bookSlots(day).map((h) => (
              <button key={h} type="button" onClick={() => onTime(h)} aria-pressed={time === h}
                className={`h-9 rounded-[10px] font-inter text-[13px] tabular-nums transition-colors ${time === h ? 'bg-ndm-deep text-white' : 'bg-white text-ndm-ink ring-1 ring-ndm-ink/10 hover:ring-ndm-deep'}`}>{bookHour(h)}</button>
            ))}
          </div>
        ) : <p className="mt-2 font-inter text-[13px] leading-[1.5] text-ndm-muted">Open days are marked with a dot. Sundays are closed.</p>}
      </div>
    </div>
  );
}
/* the three-step request form, shared by the booking section and the Book Free Consultation pop-up */
function BookForm({ thanksId, summary = false }) {
  const box = useRef(null);
  const [step, setStep] = useState(0);
  const [sent, setSent] = useState(false);
  const [err, setErr] = useState('');
  const [f, setF] = useState({ interest: '', clinic: 'Uptown', day: null, time: null, via: ['Call'], news: false, first: '' });
  const set = (k, v) => { setErr(''); setF((o) => ({ ...o, [k]: v })); };
  const toggle = (k, v) => { setErr(''); setF((o) => ({ ...o, [k]: o[k].includes(v) ? o[k].filter((x) => x !== v) : [...o[k], v] })); };
  const go = (n) => {
    setErr(''); setStep(n);
    setTimeout(() => { const l = box.current && box.current.querySelector('legend'); if (l) l.focus({ preventScroll: true }); }, 30);
  };
  const next = () => {
    if (step === 0 && !f.interest) { setErr('Please choose one option.'); return; }
    if (step === 1 && (!f.day || !f.time)) { setErr('Please pick a day and a time.'); return; }
    go(step + 1);
  };
  const submit = (e) => { e.preventDefault(); if (!f.via.length) { setErr('Please choose at least one way to reach you.'); return; } setSent(true); };
  const when = f.day && f.time ? `${bookDay(f.day)}, ${bookHour(f.time)}` : null;
  const legend = 'font-aeonik text-[clamp(21px,1.9vw,26px)] font-normal leading-[1.15] tracking-[-0.02em] text-ndm-ink outline-none';
  const label = 'mb-1.5 block font-inter text-[13px] font-semibold text-ndm-ink';
  const rows = [['Concern', f.interest], ['Clinic', BOOK_CLINIC_OPTS.find(([v]) => v === f.clinic)[1]], ['Day and time', when]];

  if (sent) return (
    <div role="status" className="flex flex-1 flex-col items-start justify-center gap-3 py-4">
      <span aria-hidden="true" className="grid size-12 place-items-center rounded-full bg-ndm-amber text-white"><IconCheckBadge /></span>
      <p id={thanksId} className="font-aeonik text-[clamp(24px,2.2vw,30px)] leading-[1.15] tracking-[-0.02em] text-ndm-ink">Request sent{f.first ? `, ${f.first}` : ''}.</p>
      <p className="max-w-[46ch] font-inter text-[14.5px] leading-[1.55] text-ndm-muted">We’ll confirm <b className="font-semibold text-ndm-ink">{when}</b> at our {f.clinic} clinic within one business day.</p>
      <button type="button" onClick={() => { setSent(false); setF((o) => ({ ...o, interest: '', day: null, time: null })); go(0); }} className="font-inter text-[14px] font-semibold text-ndm-deep hover:underline">Book another visit</button>
    </div>
  );

  const body = (
    <form ref={box} onSubmit={submit} className="flex min-w-0 flex-1 flex-col">
      <div className="grid grid-cols-3 gap-1.5" aria-hidden="true">
        {BOOK_STEPS.map((s, i) => <span key={s} className={`h-1 rounded-full transition-colors duration-500 ${i <= step ? 'bg-ndm-deep' : 'bg-ndm-ink/10'}`} />)}
      </div>
      <p className="mt-2 font-inter text-[12.5px] text-ndm-muted">Step {step + 1} of {BOOK_STEPS.length} · <span className="text-ndm-ink">{BOOK_STEPS[step]}</span></p>

      {step === 0 && (
        <fieldset className="mt-5">
          <legend tabIndex={-1} className={legend}>What would you like to address?</legend>
          <div className="mt-4 flex flex-wrap gap-1.5">
            {BOOK_INTERESTS.map((t) => <BookChip key={t} name="interest" value={t} checked={f.interest === t} onChange={() => set('interest', t)}>{t}</BookChip>)}
          </div>
        </fieldset>
      )}

      {step === 1 && (
        <fieldset className="mt-5">
          <legend tabIndex={-1} className={legend}>Where and when?</legend>
          <div className="mt-4 flex flex-wrap gap-1.5" role="group" aria-label="Clinic">
            {BOOK_CLINIC_OPTS.map(([v, t]) => <BookChip key={v} name="clinic" value={v} checked={f.clinic === v} onChange={() => set('clinic', v)}>{t}</BookChip>)}
          </div>
          <div className="mt-4"><BookCalendar day={f.day} time={f.time} onDay={(d) => setF((o) => ({ ...o, day: d, time: null }))} onTime={(h) => set('time', h)} /></div>
          <p className="mt-3 font-inter text-[12px] text-ndm-muted">Times are requests; our team confirms by phone or email.</p>
        </fieldset>
      )}

      {step === 2 && (
        <fieldset className="mt-5">
          <legend tabIndex={-1} className={legend}>How can we reach you?</legend>
          <div className="mt-4 grid gap-2.5 sm:grid-cols-2">
            <label><span className={label}>First name</span><input required name="first" autoComplete="given-name" value={f.first} onChange={(e) => set('first', e.target.value)} className={`${fieldCls} !h-11`} /></label>
            <label><span className={label}>Last name</span><input required name="last" autoComplete="family-name" className={`${fieldCls} !h-11`} /></label>
            <label><span className={label}>Mobile number</span><input required type="tel" name="phone" autoComplete="tel" className={`${fieldCls} !h-11`} /></label>
            <label><span className={label}>Email</span><input required type="email" name="email" autoComplete="email" className={`${fieldCls} !h-11`} /></label>
          </div>
          <div role="group" aria-label="Best way to reach you, choose any" className="mt-4 flex flex-wrap items-center gap-1.5">
            <span className="mr-1.5 font-inter text-[13px] font-semibold text-ndm-ink">Reach me by</span>
            {['Call', 'Text', 'Email'].map((t) => <BookChip key={t} type="checkbox" name="via" value={t} checked={f.via.includes(t)} onChange={() => toggle('via', t)}>{t}</BookChip>)}
          </div>
          <p className="mt-3 font-inter text-[12px] text-ndm-muted">Please don’t include medical details. We use your details only to arrange your visit.</p>
        </fieldset>
      )}

      {err && <p role="alert" className="mt-3 font-inter text-[13px] text-[#B4351F]">{err}</p>}
      <div className="mt-auto flex flex-wrap items-center gap-2.5 pt-6">
        {step > 0 && <Btn as="button" type="button" v="line" sm onClick={() => go(step - 1)}>Back</Btn>}
        {step < 2 ? <Btn as="button" type="button" v="ink" sm onClick={next}>Continue</Btn> : <Btn as="button" type="submit" v="book" hot>Book Free Consultation</Btn>}
      </div>
    </form>
  );
  if (!summary) return body;
  return (
    <div className="grid flex-1 gap-5 lg:grid-cols-[minmax(0,1fr)_minmax(0,260px)]">
      {body}
      <aside className="book-sum hidden flex-col rounded-[20px] p-5 text-white lg:flex" aria-label="Your appointment">
        <p className="font-inter text-[11.5px] uppercase tracking-[.16em] text-white/60">Your appointment</p>
        <dl className="mt-3 grid gap-3">
          {rows.map(([k, v]) => (
            <div key={k} className="border-b border-white/12 pb-3">
              <dt className="font-inter text-[11px] uppercase tracking-[.12em] text-white/55">{k}</dt>
              <dd className={`mt-1 font-aeonik text-[16.5px] leading-[1.3] ${v ? 'text-white' : 'tt-strong text-white/45'}`}>{v || 'Not chosen yet'}</dd>
            </div>
          ))}
        </dl>
        <p className="mt-auto pt-5 font-inter text-[12.5px] leading-[1.5] text-white/65">Prefer to talk? Call {CLINICS.map((c) => <a key={c.name} href={tel(c.phone)} className="block font-semibold text-white hover:underline">{c.name} {c.phone}</a>)}</p>
      </aside>
    </div>
  );
}
